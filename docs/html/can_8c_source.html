23

29 DDRE &= ~(1 << PE0);
30 PORTE |= (1 << PE0);
31
32 // Disable global interrupts
33 cli();
34 // Enable INT2
35 GICR |= (1 << CAN_INTERRUPT_ISR_REGISTER);
36
37 // Edge select, active-low interrupt from MCP2515
38 EMCUCR &= ~(1 << ISC2);

49int8_t can_write(struct can_device *dev, struct CAN_frame msg) {
50
51 if (msg.extended) {
52 // 29bit ID
53 uint32_t id = msg.id & 0x1FFFFFFF;
54
55 uint8_t EID0 = id & 0xFF; // EID[7:0]
56 uint8_t EID8 = (id >> 8) & 0xFF; // EID[15:8]
57 uint8_t SIDL = ((id >> 16) & 0x03) // EID[17:16] -> SIDL<1:0>
58 | 0x08 // EXIDE=1 (bit3)
59 | ((id >> 18) & 0xE0); // SID[2:0] -> SIDL<7:5>
60 uint8_t SIDH = (id >> 21) & 0xFF; // SID[10:3] -> SIDH
61
62 MCP2515_write(dev, MCP2515_TXB0SIDH, SIDH);
63 MCP2515_write(dev, MCP2515_TXB0SIDL, SIDL);
64 MCP2515_write(dev, MCP2515_TXB0EID8, EID8);
65 MCP2515_write(dev, MCP2515_TXB0EID0, EID0);
66 } else { // 11bit ID
67 uint8_t ID_MSB = (0x7F8 & msg.id) >> 3;
68 uint8_t ID_LSB = (0x7 & msg.id) << 5;
69
70 MCP2515_write(dev, MCP2515_TXB0SIDH, ID_MSB);
71 MCP2515_write(dev, MCP2515_TXB0SIDL, ID_LSB);
72 }
73 MCP2515_write(dev, TXB0DLC, msg.dlc);
74
75 uint8_t buff0_status;
76 MCP2515_read(dev, MCP2515_TXB0CTRL, &buff0_status);
77 if ((buff0_status & 0x8) != 0x8) {
78 if (msg.dlc > 8) {
79 LOG_ERR("TX Buffer0 overflow, message too large", -2);
80 return -2;
81 }
82 for (uint8_t i = 0; i < msg.dlc; i++) {
83 MCP2515_write(dev, MCP2515_TXB0D0 + i, msg.data[i]);
84 }
85
86 MCP2515_request_to_send(dev, MCP2515_RTS_TX0);
87 return 0;
88
89 } else {
90 // printf("TX Buffer0 not avaliable\r\n");
91 return -2;
92 }
93}

95int8_t can_read_rx0(struct can_device *dev, struct CAN_frame *out) {
96
97 uint8_t ID_MSB;
98 uint8_t ID_LSB;
99 uint8_t length;
100
101 MCP2515_read(dev, MCP2515_RXB0SIDH, &ID_MSB);
102 MCP2515_read(dev, MCP2515_RXB0SIDL, &ID_LSB);
103
104 /*
105 ID_LSB = (ID_LSB & 0xE0) >> 5;
106 out->id = ID_MSB << 3;
107 out->id |= (out->id & 0x7F8) | (ID_LSB & 0x7);
108 */
109
110 ID_LSB = (ID_LSB & 0xE0) >> 5;
111 out->id = (ID_MSB << 3) | (ID_LSB & 0x7);
112
113 MCP2515_read(dev, MCP2515_RXB0DLC, &length);
114 out->dlc = (length & 0xF);
115 if (out->dlc > 8) {
116 out->dlc = 8;
117 }
118
119 for (uint8_t i = 0; i < out->dlc; i++) {
120 MCP2515_read(dev, MCP2515_RXB0D0 + i, (uint8_t *)&out->data[i]);
121 }
122 // Clear RX1IF to clear buffer for next message
123 MCP2515_bit_modify(dev, MCP2515_CANINTF, MCP2515_RX0IF, 0x00); // RX0IF=bit0
124 return 0;

159int can_rxq_add(struct CAN_frame *msg) {
160
161 uint8_t next = (uint8_t)((rxq_head + 1) & (CAN_RXQ_SIZE - 1));
162 if (next == rxq_tail) {
163 return 0;
164 }
165 rxq[rxq_head] = *msg;
166 rxq_head = next;
167 return 1;
168}

170// Pull from RX buffer
171int can_rxq_pull(struct CAN_frame *out) {
172 if (rxq_tail == rxq_head) {
173 return 0;
174 }
175 *out = rxq[rxq_tail];
176 rxq_tail = (uint8_t)((rxq_tail + 1) & (CAN_RXQ_SIZE - 1));
177 return 1;
178}

216int8_t MCP2515_init(struct can_device *dev) {
217
218 while (spi_ready()) { // Confirm SPI Initialized
219 uint8_t status = spi_init();
220 }
221
222 MCP2515_reset(dev); // Send reset - command
223 uint8_t value = 0;
224 _delay_ms(10);
225
226 // Confirm that we are in correct mode after boot, and that there are no
227 // pending messages.
228 MCP2515_read(dev, 0x0E, &value);
229
230 // Ensure config mode after reset
231 if ((value & MODE_MASK) != (MODE_CONFIG)) {
232 LOG_ERR("SPI to CAN controller is not in configuration mode after reset!",
233 -1);
234 return -1;
235 }
236
237 // MCP2515_read_status(dev, &value);
238 if ((value & 0x0E) != (MCP2515_NO_IRQ)) {
239 LOG_ERR("There is an interrupt request when booting the SPI-CAN controller",
240 -2);
241 return -2;
242 }
243
244 // RTS control
245 unsigned char tx[3] = {MCP2515_WRITE, MCP2515_TXRTSCTRL, MCP2515_TXRTS_CONF};
246 spi_send_n(&dev->spi, tx, 3);
247
248 // Receive buffer0 config. Receive all, no overflow
249 MCP2515_bit_modify(dev, MCP2515_RXB0CTRL, 0x60, 0x60);
250 MCP2515_read(dev, MCP2515_RXB0CTRL, &value);
251 if (value != 0x60) {
252 LOG_ERR("Couldnt configure MCP2515 RX0 Buffer", -3);
253 return -3;
254 }
255
256 // Receive buffer1 config. Receive all
257 MCP2515_bit_modify(dev, MCP2515_RXB1CTRL, 0x60, 0x60);
258 MCP2515_read(dev, MCP2515_RXB1CTRL, &value);
259 if (value != 0x60) {
260 LOG_ERR("Couldnt configure MCP2515 RX1", -4);
261 return -4;
262 }
263
264 // Set PE0 as input
265 DDRE &= ~(1 << PE0);
266 PORTE |= (1 << PE0);
267
268 // Enable INT2
269 GICR |= (1 << INT2);
270
271 // Edge select, active-low interrupt from MCP2515
272 EMCUCR &= ~(1 << ISC2);
273
274 // Interrupt config: msg error, error flag change, TX0 empty, RX0 full
275 MCP2515_bit_modify(dev, MCP2515_CANINTE, 0xFF, MCP2515_RX_IRQ);
276 MCP2515_read(dev, MCP2515_CANINTE, &value);
277 if ((value & MCP2515_RX_IRQ) != MCP2515_RX_IRQ) {
278 LOG_ERR("Couldnt set IRQ config for MCP2515", -5);
279 return -5;
280 }
281
282 // BRP and CAN timing config
283 MCP2515_bit_modify(dev, MCP2515_CNF1, MCP2515_VAL_CNF1, MCP2515_VAL_CNF1);
284 MCP2515_bit_modify(dev, MCP2515_CNF2, MCP2515_VAL_CNF2, MCP2515_VAL_CNF2);
285 MCP2515_bit_modify(dev, MCP2515_CNF3, MCP2515_VAL_CNF3, MCP2515_VAL_CNF3);
286
287 MCP2515_read(dev, MCP2515_CNF1, &value);
288 if (value != MCP2515_VAL_CNF1) {
289 LOG_ERR("Couldnt configure BRP for MCP2515", -6);
290 return -6;
291 }
292
293 MCP2515_read(dev, MCP2515_CNF2, &value);
294 if (value != MCP2515_VAL_CNF2) {
295 LOG_ERR("CNF2 not set", -7);
296 return -7;
297 }
298
299 MCP2515_read(dev, MCP2515_CNF3, &value);
300 if (value != MCP2515_VAL_CNF3) {
301 LOG_ERR("CNF3 not set", -8);
302 return -8;
303 }
304
305 // Set the controller to normal mode !!!THIS HAS TO BE LAST IN INIT!!!
306 MCP2515_write(dev, MCP2515_CANCTRL, MODE_NORMAL);
307 MCP2515_read(dev, MCP2515_CANSTAT, &value);
308 if ((value & MODE_MASK) != MODE_NORMAL) {
309 LOG_ERR("MCP2515 could not be set to NORMAL mode when init completed", -9);
310 return -9;
311 }
312
313 return 0;
314}
