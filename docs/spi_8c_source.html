131int spi_duplex(enum spi_slave *slave, unsigned char *data, unsigned char *out,
132 int length) {
133 spi_set_slave_select(slave, 0);
134
135 for (int i = 0; i < length; i++) {
136 unsigned char tx = 0xFF;
137 if (data != TRASHCAN) {
138 tx = data[i];
139 }
140
141 SPDR = tx;
142 while (!(SPSR & (1 << SPIF))) {
143 }
144
145 if (out != TRASHCAN) {
146 out[i] = SPDR;
147 }
148 }
149
150 spi_set_slave_select(slave, 1);
