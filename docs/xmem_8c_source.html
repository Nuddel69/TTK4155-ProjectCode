16

18 // The pins in range PC7-PC4 is reserved for JTAG, and XMBK to stop pins from
19 // floating see pg32 datasheet
20 // SFIOR =
21 // (SFIOR & ~((1 << XMM1) | (1 << XMM0))) | ((1 << XMM2) /* | (1 <<
22 // XMBK)*/);
23 SFIOR |= (1 << XMM2);
24
25 // Extended MCU Control Register pg30 datasheet
26 // Setting SRL2-SRL0 as 1 gives the sectors lower sector= 0x1100-0xDFFF & high
27 // sector =0xE000-=xFFFF Setting SRW01 & SRW00 as 1 gives waitstate 2 cycles
28 // before writing and 1 cycle before sending new address, see pg 31
29 // EMCUCR =
30 // (1 << SRW01) | (1 << SRW00) & ~(1 << SRL2) & ~(1 << SRL1)6(1 << SRL0);
31
32 // EMCUCR = (1 << SRW11 | 1 << SRW10) & ~(SRL0) & ~(SRL1) & ~(SRL2);
33 // EMCUCR = 0;
34
35 // MCU Control Register pg30
36 // Same as EMCUCR but for upper sector, set to wait 1 cycle for read & write,
37 // see table at pg31
38 // MCUCR = (MCUCR & ~(1 << SRW11)) | (1 << SRW10);
39 return 0;
40}
